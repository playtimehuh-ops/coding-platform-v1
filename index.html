async function startCheckout(plan){
  if(plan==="free"){
    toast("Free is available automatically.");
    return;
  }
  if(!state.token){
    closePlans();
    openTokenModal();

    return;
  }
  const button=$("plans").querySelector('[data-plan="'+CSS.escape(plan)+'"]');
  if(button)button.disabled=true;
  try{
    const res=await fetch("/api/billing",{

  }catch(error){
    $("billingNote").textContent=error.message||"Checkout failed.";
    if(button)button.disabled=false;
  }
}

function saveActiveFile(){
  if(!state.activeFile)return;
  state.activeFile.content=$("source").innerText||$("source").textContent||"";
  state.activeFile.size=state.activeFile.content.length;
  saveWorkspace();
  renderCode();
}

function saveWorkspace(){
  try{
    localStorage.setItem("codebase-local-project",JSON.stringify({
      projectName:state.projectName,
      files:state.files
    }));
  }catch{}
}

function restoreWorkspace(){
  try{
    const raw=localStorage.getItem("codebase-local-project");
    if(!raw)return;
    const data=JSON.parse(raw);
    if(!Array.isArray(data.files)||!data.files.length)return;
    state.projectName=String(data.projectName||"Local project");
    state.files=data.files.filter(f=>f&&typeof f.path==="string"&&typeof f.content==="string");
    state.activeFile=state.files[0]||null;
    $("repoMeta").textContent=state.projectName+" · "+state.files.length+" files · local only";
    renderTree();
    renderCode();
    log("Restored local workspace.","ok");
  }catch{}
}

async function openFolder(){
  if(window.showDirectoryPicker){
    try{
      const handle=await window.showDirectoryPicker();
      const files=[];
      async function walk(dir,prefix){
        for await(const entry of dir.values()){
          const path=prefix?prefix+"/"+entry.name:entry.name;
          if(entry.kind==="directory"){
            if(["node_modules",".git","dist","build","coverage",".next"].includes(entry.name))continue;
            await walk(entry,path);
          }else if(entry.kind==="file"){
            if(!/\.(js|jsx|ts|tsx|json|html|css|scss|md|py|go|rs|java|php|rb|vue|svelte|yml|yaml|sql|sh|txt)$/i.test(entry.name))continue;
            try{
              const file=await entry.getFile();
              const text=await file.text();
              files.push({path,sha:"",size:text.length,content:text.slice(0,120000)});
            }catch{}
          }
        }
      }
      await walk(handle,"");
      state.projectName=handle.name||"Local project";
      state.files=files.slice(0,80);
      state.activeFile=state.files[0]||null;
      $("repoMeta").textContent=state.projectName+" · "+state.files.length+" files · local only";
      renderTree();
      renderCode();
      saveWorkspace();
      log("Loaded "+state.files.length+" source files from "+state.projectName+".","ok");
      return;
    }catch(error){
      if(error?.name==="AbortError")return;
      toast(error.message||"Could not open the folder.");
    }
  }
  $("folderInput").click();
}

$("folderInput").addEventListener("change",async e=>{
  const files=[];
  for(const file of Array.from(e.target.files||[]).slice(0,80)){
    if(!/\.(js|jsx|ts|tsx|json|html|css|scss|md|py|go|rs|java|php|rb|vue|svelte|yml|yaml|sql|sh|txt)$/i.test(file.name))continue;
    const text=await file.text().catch(()=> "");
    files.push({path:file.webkitRelativePath||file.name,sha:"",size:text.length,content:text.slice(0,120000)});
  }
  if(!files.length){toast("No supported source files found.");return}
  state.projectName=(e.target.files?.[0]?.webkitRelativePath||"Local project").split("/")[0]||"Local project";
  state.files=files;
  state.activeFile=state.files[0]||null;
  $("repoMeta").textContent=state.projectName+" · "+state.files.length+" files · local only";
  renderTree();
  renderCode();
  saveWorkspace();
  log("Loaded "+state.files.length+" source files.","ok");
});

function applyChanges(changes){
  for(const change of changes||[]){
    const existing=state.files.find(file=>file.path===change.path);
    if(existing){
      existing.content=String(change.content||"");
      existing.size=existing.content.length;
    }else{
      state.files.push({path:change.path,sha:"",size:String(change.content||"").length,content:String(change.content||"")});
    }
  }
  state.activeFile=state.files.find(f=>f.path===state.activeFile?.path)||state.files[0]||null;
  saveWorkspace();
  renderTree();
  renderCode();
  log("Applied "+(changes?.length||0)+" change(s) locally.","ok");
  toast("Changes applied locally.");
}

function downloadText(filename,text,mime){
  const url=URL.createObjectURL(new Blob([text],{type:mime||"text/plain"}));
  const a=document.createElement("a");
  a.href=url;a.download=filename;
  document.body.appendChild(a);a.click();a.remove();
  setTimeout(()=>URL.revokeObjectURL(url),1000);
}

function downloadPatch(changes){
  const patch=(changes||[]).map(change=>{
    const before=state.files.find(f=>f.path===change.path)?.content||"";
    return "diff --git a/"+change.path+" b/"+change.path+"\n"+simpleDiff(before,change.content);
  }).join("\n\n");
  downloadText("codebase-local.patch",patch,"text/x-diff");
}

async function send(){
  const prompt=$("prompt").value.trim();
  if(!prompt)return;
  saveActiveFile();
  const files=currentContext();
  if(!files.length){
    toast("Open a local project folder first.");
    return;
  }
  $("prompt").value="";
  addMessage("YOU",prompt);
  const thinking=addMessage("AI","Inspecting the local project and planning the change…");
  $("sendBtn").disabled=true;
  try{
    if(!state.token){
      thinking.remove();
      $("prompt").value=prompt;
      openTokenModal();
      $("sendBtn").disabled=false;
      return;
    }
    const res=await fetch("/api/chat",{
      method:"POST",
      headers:{"content-type":"application/json",...accountHeaders()},
      body:JSON.stringify({prompt,model:$("modelSelect").value||"auto:coding",context:{repository:state.projectName,branch:"local",language:"mixed",description:"Local browser workspace",files}})
    });
    const data=await res.json();
    thinking.remove();
    if(!res.ok)throw new Error(data.error||"The coding agent failed.");
    addAgentResult(data);
    log("AI proposal ready.","ok");
  }catch(error){
    thinking.remove();
    addMessage("AI",error.message||"The coding agent failed.");
    log(error.message||"Agent failed","warn");
  }finally{$("sendBtn").disabled=false}
}

function newFile(){
  const path=window.prompt("File name (for example src/app.js):");
  if(!path)return;
  if(state.files.some(f=>f.path===path)){toast("That file already exists.");return}
  state.files.push({path,sha:"",size:0,content:""});
  state.activeFile=state.files[state.files.length-1];
  saveWorkspace();
  renderTree();
  renderCode();
  log("Created "+path+" locally.","ok");
}

function exportProject(){
  saveActiveFile();
  const payload={

    exportedAt:new Date().toISOString(),

  };
  downloadText("codebase-project.json",JSON.stringify(payload,null,2),"application/json");
}
